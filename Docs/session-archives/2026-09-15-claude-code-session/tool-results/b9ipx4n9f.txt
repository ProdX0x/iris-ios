   176	/// The stages of a level, played one after the other with a breath between them.
   177	struct OculoSequenceState: Hashable, Sendable {
   178	    private(set) var stages: [OculoStageState]
   179	    let hidesLueurs: Bool
   180	    let pause: TimeInterval
   181	    private(set) var currentIndex = 0
   182	    /// While a stage just completed, the breath before the next one.
   183	    private(set) var pauseRemaining: TimeInterval = 0
   184	    private(set) var completedAt: TimeInterval?
   185	    private(set) var successes = 0
   186	    private(set) var misses = 0
   187	    /// When the current stage started: every stage runs on its own clock (0 when it begins), so a stage placed after
   188	    /// others in a sequence behaves exactly as it does alone.
   189	    private(set) var stageStartedAt: TimeInterval = 0
   190	
   191	    /// Chapter XII: a constellation of one star per stage, around `constellationCenter`.
   192	    let showsConstellation: Bool
   193	    let constellationCenter: Vector2
   194	    let constellationRadius: Double
   195	
   196	    init(stages: [OculoStageState], hidesLueurs: Bool, pause: TimeInterval, showsConstellation: Bool = false,
   197	         constellationCenter: Vector2 = .zero, constellationRadius: Double = 0) {
   198	        self.stages = stages
   199	        self.hidesLueurs = hidesLueurs
   200	        self.pause = pause
   201	        self.showsConstellation = showsConstellation
   202	        self.constellationCenter = constellationCenter
   203	        self.constellationRadius = constellationRadius
   204	        if stages.isEmpty { completedAt = 0 }
   205	    }
   206	
   207	    var isComplete: Bool { completedAt != nil }
   208	    var isBreathing: Bool { pauseRemaining > 0 }
   209	    var current: OculoStageState? { stages.indices.contains(currentIndex) ? stages[currentIndex] : nil }
   210	    var stageCount: Int { stages.count }
   211	
   212	    var progress: Double {
   213	        guard !stages.isEmpty else { return 1 }
   214	        if isComplete { return 1 }
   215	        return (Double(currentIndex) + (current?.progress ?? 0)) / Double(stages.count)
   216	    }
   217	
   218	    enum Change: Hashable, Sendable {
   219	        case success(stage: Int)
   220	        case miss(stage: Int)
   221	        case stageCompleted(stage: Int)
   222	        case completed
   223	    }
   224	
   225	    mutating func update(_ input: OculoInput, targets: [Target], braisesLit: [Int]) -> (changes: [Change], impulses: [OculoImpulse]) {
   226	        guard !isComplete, stages.indices.contains(currentIndex) else { return ([], []) }
   227	        if pauseRemaining > 0 {
   228	            pauseRemaining = max(0, pauseRemaining - input.seconds)
   229	            if pauseRemaining <= 0 { stageStartedAt = input.elapsed }
   230	            return ([], [])
   231	        }
   232	        let local = OculoInput(seconds: input.seconds, gaze: input.gaze, gazeActive: input.gazeActive, head: input.head,
   233	                               elapsed: input.elapsed - stageStartedAt)
   234	        let outcome = stages[currentIndex].update(local, targets: targets, braisesLit: braisesLit)
   235	        var changes: [Change] = []
   236	        for change in outcome.changes {
   237	            switch change {
   238	            case .success:
   239	                successes += 1
   240	                changes.append(.success(stage: currentIndex))
   241	            case .miss:
   242	                misses += 1
   243	                changes.append(.miss(stage: currentIndex))
   244	            case .completed:
   245	                changes.append(.stageCompleted(stage: currentIndex))
   246	            }
   247	        }
   248	        if stages[currentIndex].isComplete {
   249	            if !changes.contains(.stageCompleted(stage: currentIndex)) { changes.append(.stageCompleted(stage: currentIndex)) }
   250	            currentIndex += 1
   251	            if currentIndex >= stages.count {
   252	                completedAt = input.elapsed
   253	                changes.append(.completed)
   254	            } else {
   255	                pauseRemaining = pause
   256	            }
   257	        }
   258	        return (changes, outcome.impulses)
   259	    }
   260	
   261	    /// Time on the current stage's own clock.
   262	    func stageTime(at elapsed: TimeInterval) -> TimeInterval {
   263	        elapsed - stageStartedAt
   264	    }
   265	
   266	    func suggestedGaze(at elapsed: TimeInterval) -> Vector2? {
   267	        guard !isComplete, pauseRemaining <= 0 else { return nil }
   268	        return current?.suggestedGaze(at: stageTime(at: elapsed))
   269	    }
   270	
   271	    var suggestedHead: HeadPose? {
   272	        guard !isComplete, pauseRemaining <= 0 else { return nil }
   273	        return current?.suggestedHead
   274	    }
   275	}

     1	// AncreStageState.swift
     2	// Layer: GameEngine
     3	// Purpose: Chapter X final: the gaze stays on an anchor while small head turns push a notch around a compass into the
     4	// band the level designates; each band holds for a while, then the head comes back before the next one. The sign of
     5	// each axis is learned from the player's own first turn, so no head convention is assumed.
     6	
     7	import Foundation
     8	
     9	struct AncreStageState: Hashable, Sendable {
    10	    let anchor: Vector2
    11	    let radius: Double
    12	    let releaseRadius: Double
    13	    let ringRadius: Double
    14	    let bands: [AncreDefinition.Band]
    15	    let yawThreshold: Double
    16	    let pitchThreshold: Double
    17	    let yawRelease: Double
    18	    let pitchRelease: Double
    19	    let dwell: TimeInterval
    20	
    21	    private(set) var baseline: HeadPose?
    22	    /// +1 or -1: the sign of the player's first yaw turn and first pitch tilt (nil until they happen).
    23	    private(set) var yawSign: Double?
    24	    private(set) var pitchSign: Double?
    25	    private(set) var bandIndex = 0
    26	    /// After a band, the head must come back near its rest before the next band counts.
    27	    private(set) var awaitingReturn = false
    28	    private(set) var isOnAnchor = false
    29	    private(set) var inBand = false
    30	    private(set) var holdTime: TimeInterval = 0
    31	    /// Times the gaze left the anchor while a band was holding.
    32	    private(set) var slips = 0
    33	    private(set) var lastDelta: HeadPose = .neutral
    34	    private(set) var completedAt: TimeInterval?
    35	
    36	    init(definition: AncreDefinition, bounds: PlayfieldBounds, shortSide: Double) {
    37	        anchor = definition.anchor.absolute(in: bounds)
    38	        radius = definition.radius * shortSide
    39	        releaseRadius = definition.releaseRadius * shortSide
    40	        ringRadius = definition.ringRadius * shortSide
    41	        bands = definition.bands
    42	        yawThreshold = definition.yawThreshold
    43	        pitchThreshold = definition.pitchThreshold
    44	        yawRelease = definition.yawThreshold * 0.6
    45	        pitchRelease = definition.pitchThreshold * 0.6
    46	        dwell = definition.dwell
    47	        if bands.isEmpty { completedAt = 0 }
    48	    }
    49	
    50	    var isComplete: Bool { completedAt != nil }
    51	    var band: AncreDefinition.Band? { bands.indices.contains(bandIndex) ? bands[bandIndex] : nil }
    52	    var progress: Double { bands.isEmpty ? 1 : (Double(bandIndex) + (dwell > 0 ? min(1, holdTime / dwell) : 0)) / Double(bands.count) }
    53	
    54	    /// Where a band sits on the compass: the first yaw turn to the right, its opposite to the left, the first pitch
    55	    /// tilt up, its opposite down.
    56	    static func angle(of band: AncreDefinition.Band) -> Double {
    57	        switch (band.axis, band.direction) {
    58	        case (.yaw, .first), (.yaw, .same): 0
    59	        case (.yaw, .opposite): Double.pi
    60	        case (.pitch, .first), (.pitch, .same): -Double.pi / 2
    61	        case (.pitch, .opposite): Double.pi / 2
    62	        }
    63	    }
    64	
    65	    /// The head turn in compass terms: +x toward the first yaw band, -y toward the first pitch band, 1 at the threshold.
    66	    func screenOffset() -> (x: Double, y: Double) {
    67	        (lastDelta.yaw * (yawSign ?? 1) / yawThreshold, -lastDelta.pitch * (pitchSign ?? 1) / pitchThreshold)
    68	    }
    69	
    70	    private func reaches(_ delta: HeadPose, _ band: AncreDefinition.Band, threshold yaw: Double, _ pitch: Double) -> Bool {
    71	        switch (band.axis, band.direction) {
    72	        case (.yaw, .first): abs(delta.yaw) >= yaw
    73	        case (.yaw, .same): delta.yaw * (yawSign ?? 1) >= yaw
    74	        case (.yaw, .opposite): -delta.yaw * (yawSign ?? 1) >= yaw
    75	        case (.pitch, .first): abs(delta.pitch) >= pitch
    76	        case (.pitch, .same): delta.pitch * (pitchSign ?? 1) >= pitch
    77	        case (.pitch, .opposite): -delta.pitch * (pitchSign ?? 1) >= pitch
    78	        }
    79	    }
    80	
    81	    mutating func update(_ input: OculoInput) -> OculoOutcome {
    82	        var outcome = OculoOutcome()
    83	        guard !isComplete, let band, let head = input.head else { return outcome }
    84	        if baseline == nil { baseline = head }
    85	        let rest = baseline ?? head
    86	        let delta = HeadPose(yaw: head.yaw - rest.yaw, pitch: head.pitch - rest.pitch)
    87	        lastDelta = delta
    88	        if input.gazeActive {
    89	            let distance = input.gaze.distance(to: anchor)
    90	            if isOnAnchor {
    91	                if distance > releaseRadius { isOnAnchor = false }
    92	            } else if distance <= radius {
    93	                isOnAnchor = true
    94	            }
    95	        } else {
    96	            isOnAnchor = false
    97	        }
    98	        if awaitingReturn {
    99	            inBand = false
   100	            holdTime = 0
   101	            if abs(delta.yaw) < yawRelease && abs(delta.pitch) < pitchRelease { awaitingReturn = false }
   102	            return outcome
   103	        }
   104	        // The first turn on an axis sets its sign, as soon as it is clearly under way.
   105	        if band.direction == .first {
   106	            switch band.axis {
   107	            case .yaw where abs(delta.yaw) >= yawRelease: yawSign = delta.yaw >= 0 ? 1 : -1
   108	            case .pitch where abs(delta.pitch) >= pitchRelease: pitchSign = delta.pitch >= 0 ? 1 : -1
   109	            default: break
   110	            }
   111	        }
   112	        inBand = inBand ? reaches(delta, band, threshold: yawRelease, pitchRelease) : reaches(delta, band, threshold: yawThreshold, pitchThreshold)
   113	        if inBand && isOnAnchor {
   114	            holdTime += input.seconds
   115	            if holdTime >= dwell - 1e-9 {
   116	                bandIndex += 1
   117	                holdTime = 0
   118	                inBand = false
   119	                awaitingReturn = true
   120	                outcome.changes.append(.success)
   121	                if bandIndex >= bands.count {
   122	                    completedAt = input.elapsed
   123	                    outcome.changes.append(.completed)
   124	                }
   125	            }
   126	        } else {
   127	            if holdTime > 0 && inBand && !isOnAnchor {
   128	                slips += 1
   129	                outcome.changes.append(.miss)
   130	            }
   131	            holdTime = 0
   132	        }
   133	        return outcome
   134	    }
   135	
   136	    /// The ideal player: eyes on the anchor, a gentle turn a little past the threshold toward the band, then back to rest.
   137	    var suggestedHead: HeadPose? {
   138	        guard !isComplete else { return nil }
   139	        // No turn before the rest pose is known: the first head sample becomes the rest.
   140	        guard let rest = baseline else { return .neutral }
   141	        guard let band, !awaitingReturn else { return rest }
   142	        let yaw = yawSign ?? 1
   143	        let pitch = pitchSign ?? 1
   144	        switch (band.axis, band.direction) {
   145	        case (.yaw, .first), (.yaw, .same): return HeadPose(yaw: rest.yaw + yaw * (yawThreshold + 2.5), pitch: rest.pitch)
   146	        case (.yaw, .opposite): return HeadPose(yaw: rest.yaw - yaw * (yawThreshold + 2.5), pitch: rest.pitch)
   147	        case (.pitch, .first), (.pitch, .same): return HeadPose(yaw: rest.yaw, pitch: rest.pitch + pitch * (pitchThreshold + 2.5))
   148	        case (.pitch, .opposite): return HeadPose(yaw: rest.yaw, pitch: rest.pitch - pitch * (pitchThreshold + 2.5))
   149	        }
   150	    }
   151	}

     1	// OculoDefinition.swift
     2	// Layer: Domain
     3	// Purpose: OCULOMOTOR EXPANSION: a level may open with a sequence of gaze-contingent stages (one per paradigm),
     4	// each a small deterministic machine; while the sequence is incomplete the lueurs may stay latent

     6	import Foundation

     8	/// One stage of a gaze-contingent sequence. Cases are added chapter by chapter.
     9	enum OculoStageDefinition: Hashable, Sendable {
    10	    /// Chapter II: fixation stability with distractor inhibition.
    11	    case coeur(CoeurDefinition)
    12	    /// Chapter III: smooth pursuit.
    13	    case fil(FilDefinition)
    14	    /// Chapter IV: anti-saccade (inhibitory control).
    15	    case miroir(MiroirDefinition)
    16	    /// Chapter V: memory-guided saccades.
    17	    case etoiles(EtoilesDefinition)
    18	    /// Chapter VI: visual search and systematic scanning.
    19	    case jardin(JardinDefinition)
    20	    /// Chapter VII: diagonal saccades of variable amplitude.
    21	    case croisement(CroisementDefinition)
    22	    /// Chapter VIII: predictive pursuit.
    23	    case courant(CourantDefinition)
    24	    /// Chapter IX: fixation disengagement (gap and overlap shifts).
    25	    case absence(AbsenceDefinition)
    26	    /// Chapter X: gaze stabilisation while the head turns (VOR-inspired).
    27	    case ancre(AncreDefinition)
    28	    /// Chapter XI: eye-head coordination (saccade-first gaze shift).
    29	    case tourner(TournerDefinition)
    30	}

    32	/// Chapter XI, « d'abord les yeux »: braises light at the edge one after another; reaching one with the eyes first and
    33	/// letting the head follow gives all its warmth, the head before the eyes half, the eyes alone a third.
    34	struct TournerDefinition: Hashable, Sendable {
    35	    let places: [NormalizedPoint]
    36	    /// The order in which the braises light (indices into `places`, repeated as needed).
    37	    let order: [Int]
    38	    let radius: Double
    39	    let releaseRadius: Double
    40	    /// Rest on a braise that counts as reaching it.
    41	    let dwell: TimeInterval
    42	    /// Head motion (|Δyaw| + |Δpitch|, degrees) when the eyes arrive under which the head has not moved yet.
    43	    let headStill: Double
    44	    /// Head motion, after the eyes arrived, that counts as the head following.
    45	    let headTurn: Double
    46	    let followWindow: TimeInterval
    47	    let announceTimeout: TimeInterval
    48	    let pause: TimeInterval
    49	    /// Warmth needed, and the warmth each coordination gives.
    50	    let goal: Double
    51	    let fullYield: Double
    52	    let headFirstYield: Double
    53	    let eyesOnlyYield: Double
    54	
    55	    init(places: [NormalizedPoint], order: [Int], radius: Double = 0.2, releaseRadius: Double = 0.27, dwell: TimeInterval = 0.2,
    56	         headStill: Double = 3, headTurn: Double = 5, followWindow: TimeInterval = 1.5, announceTimeout: TimeInterval = 4,
    57	         pause: TimeInterval = 0.6, goal: Double = 4, fullYield: Double = 1, headFirstYield: Double = 0.5, eyesOnlyYield: Double = 0.34) {
    58	        self.places = places
    59	        self.order = order.filter { places.indices.contains($0) }
    60	        self.radius = max(radius, 0.05)
    61	        self.releaseRadius = max(releaseRadius, self.radius)
    62	        self.dwell = max(dwell, 0.05)
    63	        self.headStill = max(headStill, 0)
    64	        self.headTurn = max(headTurn, self.headStill)
    65	        self.followWindow = max(followWindow, 0.2)
    66	        self.announceTimeout = max(announceTimeout, 1)
    67	        self.pause = max(pause, 0)
    68	        self.goal = max(goal, 0)
    69	        self.fullYield = max(fullYield, 0)
    70	        self.headFirstYield = max(headFirstYield, 0)
    71	        self.eyesOnlyYield = max(eyesOnlyYield, 0.01)
    72	    }
    73	}
